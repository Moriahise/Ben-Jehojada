@@Vayeshev 9:2@@
וּבִרְכַּת שֶׁלֹּא עָשַׂנִי גּוֹי זוֹ נֻוֽסְחָה אֲמִיֽתִּית שֶׁהֱבִיאָהּ מָרָן זַ"ל בב"י מִן גְּמָרָא דִּמְנָחוֹת דַּף מ"ג, וְכֵן הָרִי"ף בְּסוֹף פ"ט דִּבְרָכוֹת הֵבִיא נֻוֽסְחָה זוֹ, וְכֵן בְּסִיֽדּוּר רַבֵּינוּ עַמְרָם גָּאוֹן וכ"כ בַּאֲגֻוֽדָּה, וּמַה שֶׁתֵּרָאֶה בְּאֵיזֶה דְּפוּסִים בְּהָרֹא"שׁ וְגַם בש"ע שֶׁעָשַׂנִי יִשְׂרָאֵל אוֹ שֶׁעָשַׂנִי יְהוּדִי זֶה הַשִּׁיֽנּוּי הוּא מִן הַמַּדְפִּיסִים, כִּי נִתְפַּחֲדוּ שֶׁלֹּא יַעֲשׂוּ עֲלֵיהֶם עֲלִילָה כַּאֲשֶׁר עָשׂוּ בִּזְמַן אֶחָד בַּוִּיֽכּוּחַ שֶׁהָיָה בֵּין הַיְּהוּדִים לַנּוֹצְרִים לְכָךְ שִׁיֽנּוּ וְלֹא הִדְפִּיסוּ שֶׁלֹּא עָשַׂנִי גּוֹי, וכ"כ הָרַב מג"א זַ"ל ס"ק ט' צ"ל שֶׁלֹּא עָשַׂנִי גּוֹי וְגִיֽרְסַת שֶׁעָשַׂנִי יְהוּדִי שִׁיֽנּוּי מִן הַמַּדְפִּיסִים ע"כ, וכ"כ הָרַב מַאֲמַר מָרְדְּכַי זַ"ל רָאִיתִי בש"ע בִּשְׁנֵי דְּפוּסִים שֶׁכָּתוּב שָׁם שֶׁעָשַׂנִי יְהוּדִי, וְנִרְאֶה דְּעִיֽקַּר הַנֻּוֽסְחָה שֶׁלֹּא עָשַׂנִי גּוֹי וּמַה שֶׁשָּׁנוּ בש"ע שֶׁעָשַׂנִי יְהוּדִי הוּא נִרְאֶה שֶׁהוּא מִפַּחַד הָאֻוֽמּוֹת, וְהָרְאָיָה שֶׁשִּׁנּוּי זֶה לֹא מְצָאנֻוֽהוּ אֶלָּא בש"ע שֶׁנִּדְפַּס בּוֹינְצְיָא וּמַנְטוֹבָה וְשִׁיֽנּוּי הַמַּדְפִּיס הוּא שֶׁהֲרֵי בְּבֵית יוֹסֵף לֹא הִזְכִּיר נֹוֽסַח זֶה כְּלָל עכ"ד יע"ש:

@@Vayeshev 9:3@@
וְדַע כִּי הַבְּרָכוֹת שֶׁלֹּא עָשַׂנִי גּוֹי וְשֶׁלֹּא עָשַׂנִי עֶבֶד וְשֶׁלֹּא עָשַׂנִי אִשָּׁה, הַכַּוָּנָה שֶׁאֲנַחְנוּ מוֹדִיעִים בָּזֶה שֶׁאֲנַחְנוּ שְׂמֵחִים וּמְרֻוֽצִּים בְּעֹוֽל מִצְוֹת וְאֵינָם מֻוֽטָּלִים עָלֵינוּ מִצַּד הַהֶכְרֵחַ, דִּלְכָךְ אֲנַחְנוּ מְבָרְכִים וּמְשַׁבְּחִים להשי"ת שֶׁהֶעְמִיס עָלֵינוּ מִצְוֹת הַרְבֵּה וְלֹא פָּטַר אוֹתָנוּ כִּבְנֵי נֹחַ אוֹ כְּעֶבֶד וְאִשָּׁה, וּבְאָוֽמְרוֹ שֶׁלֹּא עָשַׂנִי אִשָּׁה יְכַוֵּיֽן לְבָרֵךְ אֶת השי"ת שֶׁעָשְׁתוּ אִישׁ שֶׁיֵּשׁ לוֹ זָקָן וְלֹא אִשָּׁה שֶׁאֵין לָהּ הַדְרַת פָּנִים זָקֵן, כִּי הַהֶיֽכֵּר הַגָּלוּי הַמַּבְדִּיל בֵּין אִישׁ לְאִשָּׁה הוּא הַזָּקָן, וּבִבְרָכָה זוֹ שֶׁמְּכַוֵּיֽן בָּהּ בְּדָבָר זֶה נוֹתֵן כָּבוֹד לְזָקֵן וּבַשָּׂכָר זֶה יָאִירוּ בּוֹ אוֹרוֹת דִּיקְנָא עִלָּאָה קַדִּישָׁא, וְהִזָּהֵר בְּזוֹ הַכַּוָּנָה וְהִיא נְקִיָּה וְקַלָּה:

@@Vayeshev 9:4@@
וּמַה שֶׁנִּתְקְנוּ בְּרָכוֹת אֵלּוּ בִּלְשׁוֹן שְׁלִילָה הַיְיֽנוּ כְּדֵי לְהַכְנִיס בָּזֶה שְׁתֵּי מַשְׁמָעוּת, הָא' הִיא כְּדַאֲמָרָן וְהַב' הוּא ר"ל שֶׁלֹּא נִתְחַלְּפָה נִשְׁמָתֵיֽנוּ בְּעֵת הַשֵּׁיֽנָה בְּנֶפֶשׁ גּוֹי אוֹ עֶבֶד אוֹ אִשָּׁה וְלֹא נִדְבְּקוּ בְּיַחַד זע"ז:

@@Vayeshev 10:1@@
בִּרְכַּת הַמַּעֲבִיר חַבְלֵי שֵׁיֽנָה אע"ג דִּבְרָכָה זוֹ נִתְקְנָה עַל סוֹד כָּמוּס עכ"ז אִם בְּשָׁעָה שֶׁהוּא מְבָרֵךְ אוֹתָהּ גּוֹבֶרֶת עָלָיו הַשֵּׁיֽנָה וּמִתְנַמְנֵם קְצָת יַמְתִּין עַד שֶׁתַּעֲבֹוֽר שִׁיֽנְתּוֹ לְגַמְרֵי וְאָז יְבָרֵךְ אוֹתָהּ:
