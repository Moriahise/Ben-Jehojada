אַחַ"כּ יֹאמַר לְשׁוֹן רַשְׁבִּ"י בְּפָ' נֹחַ דַּף ס"ה ע"א, א"ר שִׁמְעוֹן אֲרַמִּית יְדַאי בִּצְלוֹתִין לְעֵיֽלָּא דְּכַד רֵעוּתָא עִלָּאָה לְעֵיֽלָּא וְכוּ' עַד וְעַל דָּא כּוֹלָּא קָאֵיֽם זַכָּאָה חוּלָקְהוֹן דְּצַדִּיקַיָּיֽא בְּעָלְמָא דִּין וּבְעָלְמָא דְּאָתֵי ע"כ, וְהַמַּאֲמָר הַזֶּה עָמֹוֽק עָמֹוֽק מִי יִמְצָאֶנּוּ, ובסה"ק סוֹד יְשָׁרִים הַנִּדְפָּס בְּסוֹף רַב פְּעָלִים ח"ג, כָּתַבְתִּי שָׁם בֵּיֽאוּר שְׁתֵּי שׁוּרוֹת מִן מַאֲמָר הֲנֵז' שֶׁנִּתְבָּאֲרוּ בְּדִבְרֵי רַבֵּינוּ הָאֲרִ"י זַ"ל, וְשָׁם הֵבֵאתִי בֵּיֽאוּר עַל מַאֲמָר זֶה כֻּוֽלּוֹ מִן מהרש"ך זַ"ל יע"ש:

בְּעִנְיַן שְׁתִיַּיֽת הקהוו"א קֹוֽדֶם עַמּוּד הַשַּׁחַר לְמִי שֶׁאֵין דַּעְתּוֹ מְיֻוֽשֶּׁבֶת בִּלְעֲדָהּ, וְהוּא חֲלוּשׁ הַמֶּזֶג וּבִפְרָט כְּשֶׁהוּא יוֹם תַּעֲנִית, וְהוּא חָלוּשׁ, וְלָכֵן רוֹצֶה לִשְׁתּוֹת אַחַר שֵׁיֽנָה כְּשֶׁנֵּיֽעוֹר קֹוֽדֶם עֲלוֹת הַשַּׁחַר, נִרְאֶה דְּשָׁרֵי לֵיהּ לִשְׁתּוֹת, וְאִם הוּא מְלֻוֽמָּד לִשְׁתּוֹת הקהוו"א בסוכ"ר, וְקָשֶׁה לוֹ בְּלֹא סוכ"ר נִרְאֶה דְּיָכוֹל לִשְׁתּוֹת בסוכ"ר, דְּאֵין זֶה דֶּרֶךְ גַּאֲוָה אֶלָּא בִּכְלַל רְפוּאָה שֶׁתִּתְיַיֽשֵּׁב דַּעְתּוֹ בָּהּ, וְיִתְחַזֵּק לִבּוֹ. ואע"פ שֶׁהָרַב פר"ח זַ"ל, וְנִמְשַׁךְ אַחֲרָיו הָרַב ברכ"י זַ"ל לֹא הִתִּירוּ לִשְׁתּוֹתָהּ אֶלָּא רַק בְּלֹא סוכ"ר, הִנֵּה בזה"ז שֶׁהַחֲלִישׁוּת מְצוּיָה וְגוֹבֶרֶת אֵצֶל רֻוֽבָּן שֶׁל בְּנֵי אָדָם יֵשׁ לְהַתִּיר גַּם בסוכ"ר כִּי מַה שֶׁמַּנִּיחַ בָּהּ סוכ"ר הוּא דֶּרֶךְ רְפוּאָה, כִּי בְּלֹא סוכ"ר תַּעֲשֶׂה יֹוֽבֶשׁ בְּאָדָם מִצַּד טִבְעָהּ, וְהַמְּלֻוֽמָּד לִשְׁתּוֹתָהּ בסוכ"ר אֵינוֹ יָכוֹל לִשְׁתּוֹתָהּ בְּלֹא סוכ"ר, וְיֵשׁ לִסְמֹוֽךְ בָּזֶה עַל הַמַּתִּירִים בסוכ"ר, וְהִזְכִּירָם הָרַב מִנְחַת אַהֲרֹן כְּלָל וָא"ו אוֹת ח' ע"ש אַךְ יַעֲשֶׂה תְּנַאי עַל שְׁתִיַּיֽת הקהוו"א בסוכ"ר קֹוֽדֶם שִׁיֽנָּה בְּפֵיֽרוּשׁ:
