אֵלָיו כְּבַחַי' הִתְלַבְּשׁוּת בְּכָל הַפַּרְצוּפִין עַד זו"ן שֶׁהֵם סוֹפָא דְּכָל דַּרְגִין וְהִתְלַבְּשׁוּת הַזֶּה הוּא נִכְלָל בְּשֵׁם הוי"ה ב"ה שֶׁהוּא קוֹץ הַיּוֹ"ד וְאַרְבַּע אוֹתִיּוֹת שֶׁהֵם כּוֹלְלִים מִן אוֹר א"ס דְּטָמִיר וּגְנִיז בְּכֶתֶר עַד זו"ן. לָכֵן אֲנַחְנוּ מְיַחֲדִים הַדִּיֽבּוּר וְהַבַּקָּשָׁה עַל שֵׁם הוי"ה ב"ה, אַךְ בֶּאֱמֶת הַמָּקוֹר שֶׁל הַשֶּׁפַע וְהַהַחָרָה הוּא מִן א"ס הַמַּאֲצִיל הַכֹּל, וְלוֹ לְבַדּוֹ הַכֹּחַ לְהַשְׁפִּיעַ בַּנֶּאֱצָלִים וּלְחַדֵּשׁ שֶׁפַע תָּמִיד. וְכַאֲשֶׁר אֲנַחְנוּ מְבַקְּשִׁים הַבַּקָּשׁוֹת שֶׁלָּנוּ אָז הַדְּבָרִים עוֹלִים לְאֹזֶן ז"א בְּסוֹד הַטֵּה ה' אָזְנְךָ וּשְׁמַע וּמִשָּׁם יַעֲלוּ וְיַגִּיעוּ מִדַּרְגָּא לְדַרְגָּא עַד א"ס, וְאָז מִשָּׁם תִּהְיֶה הַפְּעֻוֽלָּה שֶׁיִּמָּשֵׁךְ הַשֶּׁפַע מִן א"ס לָעַתִּיק וּמַעְתִּיק לא"א וּמִן א"א לאו"א וּמִן או"א לזו"ן וּמֵהֶם יַגִּיעַ לָנוּ. וּזְכֹוֽר תָּמִיד הַכְּלָל הַזֶּה שֶׁכָּתַב מוה"ר זלה"ה וְהוּא כְּלָל שֶׁל דָּבָר יָשִׂים הָאָדָם שֵׁם ההוי"ה נֶגֶד עֵינָיו כמ"ש שִׁוִּיתִי ה'
