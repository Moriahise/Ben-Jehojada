אַל יַעֲשֶׂה הָאָדָם תְּפִלָּתוֹ קֶבַע אֶלָּא רַחֲמִים וְתַחֲנוּנִים וְכַמְּפוֹרָשׁ בַּמִּשְׁנָה פֶּרֶק ב' דְּאָבוֹת וּכְשֶׁאַתָּה מִתְפַּלֵּל אַל תַּעֲשֶׂה תְּפִלָּתְךָ קֶבַע אֶלָּא רַחֲמִים וְתַחֲנוּנִים שֶׁנֶּאֱמַר כִּי רַחוּם וְחַנּוּן הוּא וְכוּ'. וְכָתַב רַבֵּינוּ יוֹנָה זַ"ל בְּפֵירוּשׁוֹ לְמַס' אָבוֹת הַכַּוָּנָה שֶׁיֹּאמַר הַתְּפִלָּה כְּעָנִי הַמִּתְחַנֵּן וְשׁוֹאֵל דָּבָר שֶׁהוּא צָרִיךְ, כִּי תַּחֲנוּנִים יְדַבֵּר רָשׁ וְלֹא כְּאָדָם הַשּׁוֹאֵל וְאֵינוֹ צָרִיךְ בּוֹ שֶׁאֵינוֹ מִתְחַנֵּן בְּלֵב נָמוּךְ וְלֹא בְּשִׁבְרוֹן הַנֶּפֶשׁ, כִּי בֶּאֱמֶת כָּל אִישׁ צָרִיךְ לְבַקֵּשׁ עַל עַצְמוֹ, כִּי אָדָם אֵין בָּאָרֶץ אֲשֶׁר לֹא יֶחֱטָא, גַּם לֹא תִּהְיֶה עָלָיו הַתְּפִלָּה כְּמַשָּׂא, וְלֹא יַעֲשֶׂנָּה כְּמִי שֶׁפָּרַע חוֹבוֹ וְכוּ' עכ"ל. וְהָרַב רַבִּי יוֹסֵף בֶּן נַחְמִיאַשׂ זַ"ל שֶׁהָיָה תַּלְמִידוֹ שֶׁל הָרֹא"שׁ זַ"ל כָּתַב בְּבֵיאוּרוֹ עַל אָבוֹת וְזַ"ל, וְזֶה טַעַם כְּשֶׁאַתָּה מִתְפַּלֵּל אַל תַּעַשׂ תְּפִלָּתְךָ קֶבַע אֶלָּא תַּחֲנוּנִים וּכְמוֹ שֶׁמְּפוֹרָשׁ בְּפֶרֶק תְּפִלַּת הַשַּׁחַר מַאי קֶבַע כֹּל שֶׁתְּפִלָּתוֹ דּוֹמָה עָלָיו כְּמַשּׂאוֹי, רַבָּנָן אָמְרִי כֹּל שֶׁאֵינוֹ אוֹמְרָהּ בִּלְשׁוֹן תַּחֲנוּנִים וּבְאָבוֹת דְּרַבִּי נָתָן אַל תַּעַשׂ תְּפִלָּתְךָ שִׂיחָה אֶלָּא תַּחֲנוּנִים, וְהַכַּוָּנָה לְהִתְחַנֵּן לִפְנֵי הַמָּקוֹם בְּלֵב נָמוּךְ וְכִוֵּון הַדַּעַת עכ"ל:
