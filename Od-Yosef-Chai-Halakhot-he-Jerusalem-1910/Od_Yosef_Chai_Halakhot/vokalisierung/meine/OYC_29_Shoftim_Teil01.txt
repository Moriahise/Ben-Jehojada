# Meine Fassung · Vorlage aus Nakdan (ohne Dicta-Plene-Marken) · hier korrigieren
@@Shoftim 1:1@@
אָסוּר לְהִתְיַיחֵד עִם עֶרְוָה מִן הָעֲרָיוֹת בֵּין זְקֵנָה בֵּין יַלְדָּה שֶׁדָּבָר זֶה גּוֹרֵם לְגַלּוֹת עֶרְוָה, חוּץ מֵהָאֵם עִם בְּנָהּ וְהָאָב עִם בִּתּוֹ. וְהַבַּעַל עִם אִשְׁתּוֹ נִדָּה. אֲבָל הֶחָתָן עִם כַּלָּתוֹ שֶׁפֵּרְסָה נִדָּה קוֹדֶם שֶׁיִּבְעוֹל אוֹתָהּ אָסוּר לְהִתְיַיחֵד עִמָּהּ אֶלָּא הִיא יְשֵׁנָה בֵּין הַנָּשִׁים וְהוּא בֵּין הָאֲנָשִׁים וְאִם בָּא עָלֶיהָ בִּיאָה רִאשׁוֹנָה וְאַחַ"כּ נִטְמְאָה מוּתָּר לְהִתְיַיחֵד עִמָּהּ. וּפֹה עִירֵנוּ בַּגְדָאד יע"א, נָהֲגוּ דַּאֲפִילּוּ בָּא עָלֶיהָ בִּיאָה רִאשׁוֹנָה וְנִטְמְאָה בְּדַם בְּתוּלִים אוֹ פַּרְסָה נִדָּה אַחַר בִּיאָתוֹ עָלֶיהָ אֵינוֹ מִתְיַיחֵד עִמָּהּ עַד שֶׁתִּטְבּוֹל וְיָבֹא עָלֶיהָ פַּעַם שֵׁנִית דְּאָז מִתְיַיחֵד כְּדִין בַּעַל עִם אִשְׁתּוֹ נִדָּה:

@@Shoftim 2:1@@
הָא דְּאָמְרוּ חֲזַ"ל הִיא יְשֵׁנָה בֵּין הַנָּשִׁים וְהוּא יָשֵׁן בֵּין הָאֲנָשִׁים הוּא לְעִכּוּבָא דְּתַרְתֵּי בָּעֵינַן וְלָא סַגִּי בְּהוּא יָשֵׁן בֵּין אֲנָשִׁים בִּלְבַד דְּחָיְישִׁינַן שֶׁמָּא יָבֹא לַחֶדֶר שֶׁלָּהּ וְיִשְׁכַּב עִמָּהּ, לְכָךְ בָּעֵינַן הִיא יְשֵׁינָה בֵּין הַנָּשִׁים דְּאִיכָּא שׁוֹמְרִים אֶצְלָהּ וְלָא סַגִּי בִּישָׁנָה בֵּין נָשִׁים בִּלְבַד דְּחוֹשְׁשִׁין שֶׁמָּא הִיא תָּבֹא בַּלַּיְלָה לְחֶדֶר מִשְׁכָּבוֹ, וּלְכָךְ בָּעֵינַן שׁוֹמְרִים אֶצְלוֹ ג"כ:

@@Shoftim 3:1@@
אע"ג דְּאִיכָּא דס"ל אָח עִם אֲחוֹתוֹ מוּתָּר לְהִתְיַיחֵד וכנז' בַּגְּמָרָא קי"ל לַהֲלָכָה דְּגַם אָח עִם אֲחוֹתוֹ אָסוּר לְהִתְיַיחֵד וְאֵינוֹ מוּתָּר אֶלָּא רַק אָב עִם בִּתּוֹ וְאִשָּׁה עִם בְּנָהּ וכנז' בַּפּוֹסְקִים וְכֵן עִיקָּר בִּפְרָט בְּדוֹר פָּרוּץ:

@@Shoftim 3:2@@
וְכָתַב הַגָּאוֹן ב"ח זַ"ל מַה שֶׁאָמְרוּ אָב עִם בִּתּוֹ מוּתָּר לְהִתְיַיחֵד לָאו דַּוְקָא עִם בִּתּוֹ בִּלְבַד אֶלָּא כָּל יוֹצְאֵי חֲלָצָיו דִּינָם שָׁוֶה, כִּי בַּת בִּתּוֹ וּבַת בַּת בִּתּוֹ שָׁרִי כְּמוֹ בִּתּוֹ עכ"ל. ואע"ג דְּהַגָּאוֹן ב"ח זַ"ל נָקֵיט בִּלְשׁוֹנוֹ בַּת בִּתּוֹ וּבַת בַּת בִּתּוֹ ה"ה דְּשָׁרֵי בְּבַת בְּנוֹ וּבַת בַּת בְּנוֹ דְּמַאי שְׁנָא הָא מֵהָא מֵאַחַר דִּכְלָלָא כָּיֵיל וְאָמַר כָּל יוֹצְאֵי חֲלָצָיו דִּינָן שָׁוֶה, וְהָכִי דָּיְיקִינַן מִלְּשׁוֹן הַגָּאוֹן הֲנֵז' בְּסוֹף סִי' כ"א סְעִיף ה' דִּמְדַמֵּי לְהָא דְּאָמַר שְׁמוּאֵל שֶׁכָּתַב וְזַ"ל וְכָל יוֹצְאֵי חֲלָצָיו וְכִדְמַסִּיק דְּאָמַר שְׁמוּאֵל הַכֹּל לש"ש וְכוּ' ע"ש. ואע"ג דְּהַגָּאוֹן נו"ב מַהֲדוּרָא תִּנְיָינָא אה"ע סִי' י"ח בְּאוֹתָהּ שֶׁנִּתְיַיחֲדָה עִם זְקֵינָה אֲבִי אָבִיהָ מַשְׁמַע מִדְּבָרָיו דַּעֲבַד אִיסּוּרָא בְּיִחוּד, לְפוּם רִיהֲטָא כָּתַב כֵּן, וְאִשְׁתְּמִיט מִינֵּיהּ דִּבְרֵי הַגָּאוֹן ב"ח הֲנֵז' שֶׁכָּתַב בִּפְשִׁיטוּת דִּבְכָל יוֹצְאֵי חֲלָצָיו לֵיכָּא אִיסּוּר יִחוּד:

@@Shoftim 4:1@@
כְּשֶׁאֵירַע מַעֲשֵׂה אַמְנוֹן עִם תָּמָר. גָּזְרוּ דָּוִד הע"ה וּבֵית דִּינוֹ עַל יִחוּד פְּנוּיָה דאע"פ שֶׁאֵינָהּ עֶרְוָה בִּכְלַל יִחוּד עֲרָיוֹת הִיא. וְהִלֵּל וְשַׁמַּאי גָּזְרוּ עַל יִחוּד גְּוִיָּה. נִמְצָא כָּל הַמִּתְיַיחֵד עִם אִשָּׁה שֶׁאָסוּר לְהִתְיַיחֵד עִמָּהּ בֵּין יִשְׂרָאֵל בֵּין גּוֹיָה מַכִּין אוֹתָם מַכַּת מַרְדּוּת הָאִישׁ וְהָאִשָּׁה וּמַכְרִיזִין עֲלֵיהֶם, חוּץ מֵאֵשֶׁת אִישׁ דאע"פ שֶׁאָסוּר לְהִתְיַיחֵד עִמָּהּ אִם נִתְיַיחֲדוּ אֵין לוֹקִין כְּדֵי שֶׁלֹּא לְהוֹצִיא עָלֶיהָ לַעַז שֶׁזִּינְּתָה וְנִמְצְאוּ מוֹצִיאִין לַעַז עַל בָּנֶיהָ שֶׁהֵם מַמְזֵרִים:

@@Shoftim 5:1@@
כָּל אִשָּׁה שֶׁאָסוּר לְהִתְיַיחֵד עִמָּהּ, הַדִּין הוּא דְּאִם הָיְתָה אִשְׁתּוֹ עִמּוֹ הֲרֵי הוּא מוּתָּר לְהִתְיַיחֵד מִפְּנֵי שֶׁאִשְׁתּוֹ מְשַׁמַּרְתּוֹ. שֶׁבְּוַדַּאי לֹא יִזְנֶה בִּפְנֵי אִשְׁתּוֹ. וְכֵן הָאִשָּׁה שֶׁבַּעְלָהּ עִמָּהּ מוּתֶּרֶת לְהִתְיַחֵד עִם הָאִישׁ. אַךְ עִם הַגּוֹי אֲסוּרָה לְהִתְיַיחֵד אע"פ שֶׁהַגּוֹי אִשְׁתּוֹ עִמּוֹ מִפְּנֵי שֶׁאֵין הַגּוֹי בּוֹשׁ לִזְנוּת בִּפְנֵי אִשְׁתּוֹ:

@@Shoftim 6:1@@
אע"ג דְּיֵשׁ אוֹמְרִים אִם בַּעְלָהּ בָּעִיר אע"פ שֶׁאֵינוֹ עִמָּהּ בַּבַּיִת אֵין חוֹשְׁשִׁין לָהּ מִשּׁוּם יִחוּד מ"מ לְדַעַת רַשִׁ"י זַ"ל וּדְעִמֵּיהּ אִיסּוּרָא מִיהָא אִיכָּא רַק דְּאֵין לוֹקִין עַל זֶה. מִיהוּ עִם אִשָּׁה שֶׁלִּבּוֹ גַּס בָּהּ שֶׁהוּא מְגוּדָּל עִמָּהּ, אוֹ שֶׁהוּא קְרוֹבָה גַּם לְדַעַת הַמְּקִילִּין אָסוּר:

@@Shoftim 6:2@@
וְכֵן אִם קִינֵּא לָהּ בַּעְלָהּ שֶׁלֹּא תִּתְיַיחֵד עִם זֶה אָסוּר לְהִתְיַיחֵד עִמּוֹ לכ"ע אַף אִם בַּעֲלָהּ בָּעִיר:

@@Shoftim 7:1@@
כָּתַב בִּנְתִיבוֹת הַשָּׁלוֹם נָתִיב ט' אוֹת ז' אִשָּׁה אַחַת לֹא תִּתְיַיחֵד עִם אֲנָשִׁים הַרְבֵּה וַאֲפִילּוּ בִּכְשֵׁרִים אָסוּר עַד שֶׁתִּהְיֶה שָׁם אִשְׁתּוֹ שֶׁל אֶחָד מֵהֶם וה"ה שְׁתֵּי נָשִׁים עִם אֲנָשִׁים הַרְבֵּה. אֲבָל בִּשְׁלֹשָׁה נָשִׁים וּשְׁלֹשָׁה אֲנָשִׁים לֵית לַן בַּהּ:

@@Shoftim 7:2@@
וְיֵשׁ אוֹמְרִים דְּאִם הֵם כְּשֵׁרִים מוּתָּרִים שְׁנֵי אֲנָשִׁים לְהִתְיַחֵד עִם אִשָּׁה אַחַת וּסְתָם אֲנָשִׁים כְּשֵׁרִים הֵם. וְדַוְקָא בַּיּוֹם וּבְעִיר. אֲבָל בַּלַּיְלָה אוֹ בַּשָּׂדֶה אֲפִילּוּ בַּיּוֹם אֵין הֶיתֵּר אֶלָּא בִּשְׁלֹשָׁה כְּשֵׁרִים. ואע"ג דְּיֵשׁ מְחַלְּקִים בֵּין הוֹלְכִים לְשָׁמְרָהּ וּבֵין אֵין הוֹלְכִים לְשָׁמְרָהּ הִנֵּה הָעִיקָּר לְעִנְיַן דִּינָא כִּסְבָרַת הָאַחֲרוֹנִים זַ"ל הַחוֹלְקִים וס"ל דְּאֵין חִילּוּק בֵּין הוֹלְכִים לְשָׁמְרָהּ אוֹ לָאו. וּבִפְרָט בַּלַּיְלָה דלכ"ע צָרִיךְ שְׁלֹשָׁה כְּשֵׁרִים. יַעַן כִּי בִּשְׁנַיִם יֵשׁ לָחוּשׁ שֶׁמָּא יָשָׁן אֶחָד מֵהֶם וְתִתְיַחֵד עִם הַשֵּׁנִי. וְכָל זֶה בִּכְשֵׁרִים אֲבָל בִּפְרוּצִים אֲפִילּוּ עִם עֲשָׂרָה אֲסוּרָה לכ"ע:

@@Shoftim 8:1@@
