אַחַר שֶׁיְּגַמֵּיר עוֹשֶׂה שָׁלוֹם וְהַפְּסִיעוֹת יֹאמַר יהר"מ ה' או"א שֶׁתִּבָּנֶה בֵּית הַמִּקְדָּשׁ בִּמְהֵרָה בְּיָמֵינוּ וְנַקְרִיב לְפָנֶיךָ תְּמִידִין כְּסִדְרָן וּמוּסָפִין כְּהִלְכָתָן. ונ"ל שֶׁיְּכַוֵּין בָּזֶה ע"פ מִדְרַשׁ רז"ל שֶׁהֱבִיאוֹ הָרַב בְּנֵי יִשָּׂשכָר שֶׁאָמְרוּ אַחַר שֶׁיִּבָּנֶה בהמ"ק בב"א יַקְרִיבוּ בּוֹ כָּל תְּמִידִין וּמוּסָפִין שֶׁל שַׁבָּתוֹת וְיָמִים טוֹבִים וְרָאשֵׁי חֳדָשִׁים שֶׁלֹּא הִקְרִיבוּ יִשְׂרָאֵל בִּהְיוֹתָם בַּגָּלוּת בְּחֻרְבָּן, וּבָזֶה הַמִּדְרָשׁ עָשָׂה יִשּׁוּב הָרַב הֲנֵז' עַל מַה שֶּׁאוֹמְרִים בְּיַעֲלֶה וְיָבֹא חַג הַסּוּכּוֹת הַזֶּה וְכֵן חַג הַמַּצּוֹת הַזֶּה וְכֵן בְּחַג הַשָּׁבוּעוֹת וְיוֹם הַזִּכָּרוֹן וּשְׁמִינִי עֲצֶרֶת ור"ח, דְּבָזֶה נִתְיַישְּׁבָה מִלַּת הַזֶּה וְעַל כֵּן בְּבַקָּשָׁה זוֹ שֶׁל תְּמִידִין וּמוּסָפִין יְכַוֵּין עַל זֶה:

כָּתַב רַבֵּינוּ הַמְּאִירִי זַ"ל בְּמַסֶּכֶת מְגִילָּה דַּף ט"ז ע"א וְזַ"ל, בְּרָכוֹת שֶׁל תְּפִלָּה עֶזְרָא הַסּוֹפֵר וּבֵית דִּינוֹ שֶׁהֵם אַנְשֵׁי כְּנֶסֶת הַגְּדוֹלָה תִּקְּנוּם, וְאַף סִידּוּר שֶׁבָּהֶם עַל פִּיהֶם נַעֲשָׂה מֵהֶם, שֶׁיָּצָא לָהֶם הַסִּידּוּר מִכֹּחַ סְבָרָתָם עַל מַה שֶׁנִּרְאֶה לָהֶם לִסְמוֹךְ עִנְיָן אֶחָד לַחֲבֵירוֹ. וּמֵהֶם שֶׁיָּצָא לָהֶם הַסִּדּוּר מִכֹּחַ רֶמֶז הַמִּקְרָאוֹת הַסְּמוּכִים זֶה לָזֶה כְּמוֹ שֶׁנִּתְבָּאֵר בַּגְּמָרָא, וּמֵאַחַר שֶׁכֵּן הוּא אֵין לָנוּ לְשַׁנּוֹת סִדּוּרָם עכ"ל:
