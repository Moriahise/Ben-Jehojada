# Meine Fassung · Vorlage aus Nakdan (ohne Dicta-Plene-Marken) · hier korrigieren
@@Vayikra 1:1@@
כָּתַב מָרָן זַ"ל בְּסִי' קנ"ג סְעִיף כ"א אֵין לִקְנוֹת מְעִילִים שֶׁנִּשְׁתַּמֵּשׁ בָּהֶם הֶדְיוֹט לְתַשְׁמִישׁ קְדוּשָּׁה וְכָתַב הַגָּאוֹן מחב"ר סִי' קנ"ג אוֹת ח' דְּהָרַב מוֹר וּקְצִיעָה וְהָרַב בְּאֵר יַעֲקֹב הִתִּירוּ לַעֲשׂוֹת מִבֶּגֶד אִשָּׁה ג"כ ע"ש וְכָתַב בְּבִרְכֵּי יוֹסֵף סִי' קנ"ג אוֹת ט' בְּשֵׁם הָרַב בְּנֵי יְאוּדֶּה דְּמוּתָּר אִם שִׁינָּה צוּרָתוֹ הָרִאשֹׁנָה וְנַעֲשָׂה צוּרָה אַחֶרֶת בְּשִׁינּוּי דְּלָא הָדַר, וְעַיֵּין מ"ש בְּשֵׁם הָרַב מָהָרִ"י פִּרְאֵנוּ זַ"ל דְּלָא הִתִּיר אֶלָּא בְּתַכְשִׁיטֵי נָשִׁים שֶׁלֹּא נִשְׁתַּמֵּשׁ בָּהֶם עֲדַיִין ע"ש:

@@Vayikra 1:2@@
וְכָתַב מור"ם זַ"ל בְּסִי' קט"ז אֵין לַעֲשׂוֹת מַפּוֹת מִדְּבָרִים יְשָׁנִים שֶׁנַּעֲשָׂה בָּהֶם דָּבָר אַחֵר לְצוֹרֶךְ הֶדְיוֹט ע"ש, והט"ז הִתִּיר בְּפָרוֹכֶת שֶׁהִיא לְתַשְׁמִישׁ דְּתַשְׁמִישׁ ע"ש. וְהַגָּאוֹן חַוּוֹת יָאִיר זַ"ל סִי' קס"א הִתִּיר בִּמְעִיל לַעֲשׂוֹת מִבִּגְדֵי אִשָּׁה כֵּיוָן דְּצָרִיךְ לְשַׁנּוֹת הַמַּלְבּוּשׁ וּלְחָתְכוּ וְהָוֵי שִׁינּוּי שֶׁאֵינוֹ חוֹזֵר לִבְרִיָּיתוֹ הֵן בְּשִׁינּוּי שֵׁם הֵן בְּשִׁינּוּי מַעֲשֶׂה שֶׁנֶּחְתַּךְ ע"ש. ואע"ג דהמג"א זַ"ל הֵבִיא בְּשֵׁם מָהָרִי"ל זַ"ל דְּאַף פָּרוֹכֶת אֵין לַעֲשׂוֹת מִבְּגָדִים יְשָׁנִים, מ"מ נָהֲגוּ הָעוֹלָם הֶיתֵּר בָּזֶה מִשּׁוּם דְּשִׁינָּה צוּרָתָן וְעָשָׂה מֵהֶם כְּלִי אַחֵר ע"ש, וּפֹה עֲיָרֵינוּ בַּגְדָאד נָהֲגוּ הֶיתֵּר לַעֲשׂוֹת פָּרוֹכֶת מִבִּגְדֵי אִשָּׁה כֵּיוָן דְּמִשְׁתַּנֶּה הַבֶּגֶד בִּשְׁמוֹ דַּעְתָּהּ נִקְרָא פָּרוֹכֶת וְגַם בְּצוּרָתוֹ שֶׁנֶּחְתַּךְ חִיתּוּךְ אַחֵר חָדָשׁ:

@@Vayikra 1:3@@
וּלְעִנְיַן הֲלָכָה הַעֲלֵתִי בסה"ק מִקְבַּצְיָאל דְּהַפָּרוֹכֶת מוּתָּר לַעֲשׂוֹתָהּ מִבִּגְדֵי נָשִׁים כֵּיוָן דְּנִשְׁתַּנּוּ כָּאָמוּר אֲבָל מְעִילִים לס"ת אֵין לַעֲשׂוֹת מִבִּגְדֵי נָשִׁים אע"פ שֶׁנִּשְׁתַּנּוּ:

@@Vayikra 2:1@@
כָּתַב הָרַב פֶּתַח הַדְּבִיר זַ"ל בְּסִי' קנ"ד אוֹת וָא"ו מַפָּה שֶׁל ס"ת מְרוּקֶּמֶת כֶּסֶף שִׁבָּלְתָּהּ וְאֵינָהּ רְאוּיָה לִכְלוּם שָׁרִי לְשָׂורְפָהּ כְּדֵי לַעֲשׂוֹת מִמֶּנָּה דָּבָר אַחֵר לס"ת, כֵּן כָּתַב הָרַב בַּעַל ס' פְּרִי הָעֵץ בִּתְשׁוּבָה כ"י אֲבָל הרה"ג חק"ל במ"ב שָׁם בְּיוֹ"ד חָלַק עָלָיו וְהֶעֱלָה לֶאֱסוֹר דְּתַשְׁמִישֵׁי קְדוּשָּׁה אָסוּר לְשָׂורְפָם. וַהֲרֵי עִיקַּר הַמַּפָּה שֶׁמִּתְלַבֵּשׁ בָּהּ הס"ת כְּשֶׁשּׂוֹרְפָהּ הוֹלֶכֶת לַצִּיבּוּר וְהָרִיקּוּם שֶׁבָּהּ אֵינוֹ אֶלָּא נוֹי לַמַּפָּה, ובסו"ד תָּמַהּ עַל הָרַב חַוּוֹת יָאִיר זַ"ל סִי' קס"ב שֶׁכָּתַב דְּתַשְׁמִישֵׁי קְדוּשָּׁה שֶׁנִּתְקַלְקְלוּ לְגַמְרֵי וא"א לְהִשְׁתַּמֵּשׁ בָּהֶן עוֹד אָזְלָא קְדוּשְׁתַּיְיהוּ ע"ש דְּהוּא הֶפֶךְ פְּסַק הָרַמְבַּ"ם ע"ש:

@@Vayikra 2:2@@
וְעֵצִים שֶׁל מַעֲקֶה הַנִּקְרָא תרבזי"ן שֶׁיֵּשׁ בַּגַּג שֶׁל בֵּית הַכְּנֶסֶת וְהַצִּיבּוּר מִתְפַּלְּלִים שָׁם בִּימֵי הַקַּיִץ וְנִתְיַישְּׁנוּ הַרְבֵּה וְנִשְׁתַּבְּרוּ שֶׁצָּרִיךְ לַעֲשׂוֹת בִּמְקוֹמָם חֲדָשִׁים הוֹרֵינוּ לָהֶם לִמְכּוֹר אוֹתָם לְהַסָּקָה:

@@Vayikra 3:1@@
כָּתְבוּ הַגְּאוֹנִים בְּסֵפֶר מָגֵן גִּבּוֹרִים סִי' קנ"ב בְּאֶלֶף הַמָּגֵן ס"ק וָא"ו תֵּיבוֹת יְשָׁנוֹת שֶׁבְּבֵית הַכְּנֶסֶת שֶׁאֵינָם רְאוּיָם לְתַשְׁמִישָׁן פָּקַע קְדוּשָּׁתָן וְעַל צַד הַיּוֹתֵר טוֹב יִמְכְּרוּ לְעֵצִים לְהַסָּקָה, וּבְדָמִים יִקְנוּ חֲדָשִׁים כ"כ מחב"ר בְּשֵׁם זֶרַע אֱמֶת:

@@Vayikra 4:1@@
אֲרוֹן הַקּוֹדֶשׁ שֶׁעוֹשִׂים בְּבִנְיַן הַכּוֹתֶל בְּעַצְמוֹ שֶׁמַּנִּיחִים שָׁם ס"ת לְכָבוֹד דַּוְקָא וְאֵין מַנִּיחִים הס"ת שָׁם לַעֲמִידָה וְרוֹצִים לְהַרְחִיב לְבֵית הַכְּנֶסֶת וְלָכֵן רוֹצִים לִסְתּוֹר הַכּוֹתֶל שֶׁבָּנוּי בּוֹ אֲרוֹן הַקּוֹדֶשׁ וְיִבְנוּ אֲרוֹן הַקּוֹדֶשׁ בַּכּוֹתֶל שֶׁבּוֹנִים מֵחָדָשׁ אַחֲרָיו שֶׁל כּוֹתֶל הָרִאשׁוֹן וּמִמֵּילָא יִהְיֶה מְקוֹם כּוֹתֶל הָרִאשׁוֹן בִּכְלַל קַרְקַע שֶׁל בֵּית הַכְּנֶסֶת אָסוּר לַעֲשׂוֹת כֵּן לְהוֹרִיד מָקוֹם זֶה שֶׁל אֲרוֹן הַקּוֹדֶשׁ מִן קְדוּשָּׁתוֹ. ואע"ג דְּאֵין מַנִּיחִים הס"ת בְּלֹא תִּיק בְּתוֹךְ הָאָרוֹן בְּשׁוּם פַּעַם אַפֵּ"ה חֲשִׁיב הָאָרוֹן תַּשְׁמִישׁ קְדוּשָּׁה לְדַעַת הָרַמְבַּ"ם וְהַטּוּר וּמָרָן זַ"ל וְכַאֲשֶׁר הַעֲלֵתִי בסה"ק רַב פְּעָלִים ח"ב סִי' ך' באו"ח ע"ש:

@@Vayikra 4:2@@
אָמְנָם הַעֲלֵתִי בס"ד שָׁם בְּסִי' כ"א דְּאִם עָבְרוּ וְסָתְרוּ אֶת כּוֹתֶל הַמַּעֲרָבִי שֶׁבּוֹ אֲרוֹן הַקּוֹדֶשׁ כּוּלּוֹ וְלֹא נִשְׁאַר אֶלָּא רַק הַקַּרְקַע אָז מוּתָּר לַעֲשׂוֹת כְּחֶפְצָם וּרְצוֹנָם יע"ש:

@@Vayikra 5:1@@
מִדְרַשׁ תַּלְמוּד תּוֹרָה שֶׁלּוֹמְדִים בּוֹ תִּינוֹקוֹת שֶׁל בֵּית רַבָּן מִן אבג"ד עַד לִימּוּד דִּבְרֵי רַבּוֹתֵינוּ עַד וְעַד בִּכְלָל, וְגַם בֵּית הַמִּדְרָשׁ הַזֶּה הוּא מְקוֹם תְּפִלָּה שֶׁמִּתְפַּלְּלִין בּוֹ הַקָּהָל שמ"ע בְּכָל יוֹם וְשַׁבָּתוֹת ויו"ט הַעֲלֵתִי בסה"ק רַב פְּעָלִים ח"ב סִי' כ"ב דְּאָסוּר לְלַמֵּד שָׁם לְתִנּוֹקוֹת הַכְּתִיבָה שֶׁל עַרְבִי וְהֶגְיוֹן עַרְבִי וְגַם אָסוּר לְלַמְּדָם שָׁם בְּמִכְתָּבִים שֶׁל סוֹחֲרִים וְהֶגְיוֹנִי עַרְבִי כְּדֵי לְהִתְלַמֵּד בָּזֶה סֵדֶר הַלָּשׁוֹן וְצַחוּתוֹ אֶלָּא יַעֲשׂוּ לָהֶם חֲדָרִים אֵצֶל בהמ"ד הֲנֵז' שֶׁלֹּא נִתְקַדְּשׁוּ וִילַמְּדוּם בְּתוֹכָם לִימּוּדִים הנז"ל וְשָׁם הֵבֵאתִי דִּבְרֵי רַבֵּינוּ הַאי גָּאוֹן בִּתְשׁוּבָה שֶׁהוּבְאָה בְּסֵפֶר גְּדוֹלָה לְיִשְׂרָאֵל וּפִלְפַּלְתִּי קְצָת בָּזֶה וְאָסַרְתִּי מִדִּינָא וכ"כ בְּסֵפֶר וְיֹאמַר יִצְחָק שֶׁהֵבֵאתִי דְּבָרָיו שָׁם יע"ש:

@@Vayikra 6:1@@
בְּעִנְיַן אִיסּוּר שֶׁל נְתִיצָה בְּבֵית הַכְּנֶסֶת אִם יֵשׁ לֶאֱסוֹר בָּזֶה אֲפִילּוּ בְּבִנְיָן שֶׁנִּבְנָה מִתְּחִלָּה לְצוֹרֶךְ אָדָם וְאַחַ"כּ נַעֲשָׂה בֵּית הַכְּנֶסֶת, או"ד בכה"ג אֵין כָּאן לָאו דִּנְתִיצָה, הֵבֵאתִי סָפֵק זֶה בסה"ק רַב פְּעָלִים ח"ב סִי' כ"ג וְהֵבֵאתִי דִּבְרֵי הָרַב מַחֲנֵה חַיִּים זַ"ל סִי' כ"ז דַּף מ"ז שֶׁעָשָׂה סְבָרָה חֲדָשָׁה בד"ז, וְלַח סָמַכְתִּי עַל זֶה וְהַעֲלֵתִי לְהַחְמִיר לֶאֱסוֹר גַּם בָּזֶה יע"ש:

@@Vayikra 7:1@@
ת"ח שֶׁלּוֹמֵד כָּל הַיּוֹם בבהמ"ד וְגַם הוּא מִסְתַּחֵר בַּסְּפָרִים לְהַרְוִיחַ בַּעֲבוּר פַּרְנָסָתוֹ שֶׁאֵינוֹ לוֹקֵחַ פְּרָס מִן קוּפַּת הַקָּהָל וְלֹא מִן אִישׁ יְחִידִי וְרוֹצֶה לַעֲשׂוֹת מזא"ד לִמְכּוֹר סְפָרִים בְּקִבּוּץ אֲנָשִׁים הַנִּקְבָּצִים שָׁם לִקְנוֹת פַּעַם אַחַת בַּחוֹדֶשׁ אוֹ פַּעַם אַחַת בִּשְׁנֵי חֳדָשִׁים אוֹ יוֹתֵר אִי אֲרִיךְ לְמֶעְבַּד הָכִי לִמְכּוֹר שָׁם סְפָרִים בְּהַכְרָזָה שֶׁקּוֹרִין מזא"ד בִּשְׁבִיל שֶׁשָּׁם נִמְצָאִים אֲנָשִׁים הַרְבֵּה, כִּי הַמָּקוֹם הֲנֵז' עוֹמֵד עַל אֵם הַדֶּרֶךְ וְקָהָל שֶׁל שָׁלֹשׁ וְאַרְבַּע בָּתֵּי כְּנֵסִיּוֹת עוֹבְרִים דֶּרֶךְ שָׁם וּבָזֶה יִהְיֶה הַרְוָוחָה לְאוֹתוֹ ת"ח, הַעֲלֵתִי בְּרַב פְּעָלִים ח"ב סִי' כ"ד לְהַתִּיר בָּזֶה:
