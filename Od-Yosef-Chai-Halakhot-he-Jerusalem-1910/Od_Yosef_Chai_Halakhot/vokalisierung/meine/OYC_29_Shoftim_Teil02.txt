# Meine Fassung · Vorlage aus Nakdan (ohne Dicta-Plene-Marken) · hier korrigieren
@@Shoftim 28:1@@
אֵין לְךָ עֶרְוָה בְּחֶנֶק אֶלָּא אֵשֶׁת אִישׁ בִּלְבַד, שֶׁנֶּאֱמַר מוֹת יוּמַת הַנּוֹאֵף וְהַנּוֹאָפֶת וּמִיתָה הָאֲמוּרָה בַּתּוֹרָה סְתָם הִיא חֶנֶק. וְאִם הָיְתָה בַּת הִיא כֹּהֵן בִּשְׂרֵיפָה וּבוֹעֲלָהּ בְּחֶנֶק שֶׁנֶּאֱמַר וּבַת אִישׁ כֹּהֵן כִּי תֵחֵל לִזְנוֹת בָּאֵשׁ תִּשָּׂרֵף. וְאִם הָיְתָה נַעֲרָה מְאוֹרָסָה שְׁנֵיהֶם בִּסְקִילָה שֶׁנֶּאֱמַר כִּי תִהְיֶה נַעֲרָה בְּתוּלָה וְכוּ' וּסְקַלְתֶּם אוֹתָם בַּאֲבָנִים. וְכָל מָקוֹם שֶׁנֶּאֱמַר בַּתּוֹרָה מוֹת יוּמָתוּ דְּמֵיהֶם בָּם הֲרֵי הֵן בִּסְקִילָה:

@@Shoftim 29:1@@
שְׁאָר הָעֲרָיוֹת כּוּלָּן בְּכָרֵת בִּלְבַד וְאֵין בָּהֶם מִיתַת בֵּית דִּין. לְפִיכָךְ אִם הָיוּ שָׁם עֵדִים וְהַתְרָאָה. בֵּית דִּין מַלְקִין אוֹתָם שֶׁכָּל חַיָּיבֵי כָּרֵיתוֹת לוֹקִין:

@@Shoftim 30:1@@
הַבָּא עַל אַחַת מֵחַיָּיבֵי לָאוִין בְּמֵזִיד הוּא לוֹקֶה וְהִיא לוֹקֶה. וְאִם בְּשׁוֹגֵג פְּטוּרִין מִכְּלוּם. וְהַבָּא עַל אַחַת מֵהַשְּׁנִיּוֹת בְּמֵזִיד מַכִּין אוֹתוֹ מַכַּת מַרְדּוּת מִדִּבְרֵיהֶם. אֲבָל הַבָּא עַל אַחַת מֵחַיָּיבֵי עֲשֵׂה אֵינוֹ לוֹקֶה. וְאִם הִכּוּ אוֹתָם בֵּית דִּין מַכַּת מַרְדּוּת כְּדֵי לְהַרְחִיק מִן הָעֲבֵירָה הָרְשׁוּת בְּיָדָם:

@@Shoftim 31:1@@
יֵשׁ עֲרָיוֹת אֲסוּרוֹת מִן הַתּוֹרָה שֶׁהֵם מְפוֹרָשׁוֹת לְהֶדְיָא בַּתּוֹרָה. וְיֵשׁ שֶׁאֲסוּרוֹת מִן הַתּוֹרָה וְאֵינָם מְפוֹרָשׁוֹת אֶלָּא שֶׁנִּלְמְדוּ בְּקַל וָחוֹמֶר וּגְזֵרָה שָׁוָה. וְיֵשׁ עֲרָיוֹת אֲסוּרוֹת מִדִּבְרֵי סוֹפְרִים שֶׁהַחֲכָמִים זַ"ל אֲסָרוּם וְגָזְרוּ בָּהֶם אִיסּוּר עֶרְוָה כְּדֵי שֶׁלֹּא יִלְמְדוּ מֵהֶן לְהַתִּיר עֲרָיוֹת שֶׁהֵם אִיסּוּר דְּאוֹרָיְיתָא. וְאֵלּוּ נִקְרָאִים בִּלְשׁוֹן חֲזַ"ל בַּתַּלְמוּד בְּשֵׁם שְׁנִיּוֹת לַעֲרָיוֹת כְּלוֹמַר הַחֲכָמִים גָּזְרוּ עֲלֵיהֶם וְהֵם שְׁנִיּוֹת לַעֲרָיוֹת דְּאוֹרָיְיתָא:

@@Shoftim 31:2@@
וְהֵא לְךָ לְשׁוֹן הָרַמְבַּ"ם זַ"ל בָּהּ' אִישׁוּת פֶּרֶק א' הֲלָכָה ה' כָּל שֶׁאָסַר בִּיאָתוֹ בַּתּוֹרָה וְחַיָּיב טַל בִּיאָתוֹ כָּרֵת וְהֵם הָאֲמוּרוֹת בְּפָרָשַׁת אַחֲרֵי מוֹת הֵן הַנִּקְרָאִים עֲרָיוֹת. וְכָל אַחַת מֵהֶן נִקְרֵאת עֶרְוָה כְּגוֹן אֵם וְאָחוֹת וּבַת וְכַיּוֹצֵא בָּהֶן:

@@Shoftim 31:3@@
וְיֵשׁ נָשִׁים אֲחֵרוֹת שֶׁהֵן אֲסוּרוֹת מִפִּי הַקַּבָּלָה. וְאִיסּוּרָן מִדִּבְרֵי סוֹפְרִים וְהֵן הַנִּקְרָאוֹת שְׁנִיּוֹת, מִפְּנֵי שֶׁהֵן שְׁנִיּוֹת לָעֲרָיוֹת וְכָל אַחַת מֵהֶן נִקְרֵאת שְׁנִיָּיה. וְעֶשְׂרִים נָשִׁים הֵן וְאֵלּוּ הֵן וְכוּ' וּמְנָאָם הָרַמְבָּ"ם שָׁם אַחַת לְאַחַת כנז' וְיֵשׁ בִּפְרָטִים שֶׁל הַשְּׁנִיּוֹת דֵּעוֹת שׁוֹנִים בְּסִפְרֵי הַפּוֹסְקִים וְלֹא פֵּרַשְׁתִּי אוֹתָם כָּאן בִּשְׁבִיל הָאֲרִיכוּת וְעַיֵּין בש"ע אה"ע סִי' ט"ו וּבָאַחֲרוֹנִים שָׁם:

@@Shoftim 32:1@@
כָּל שֶׁאָסַר בִּיאָתוֹ בַּתּוֹרָה וְלֹא חִיֵּיב עָלָיו כָּרֵת הֵן הַנִּקְרָאִים אִיסּוּרֵי לָאוְיָן. וְעוֹד נִקְרָאִים אִיסּוּרֵי קְדוּשָּׁה. וְתִשְׁעָה הֵן וְאֵלּוּ הֵן. אַלְמָנָה לְכֹהֵן גָּדוֹל. גְּרוּשָׁה אוֹ זוֹנָה אוֹ חֲלָלָה בֵּין לְכֹהֵן גָּדוֹל בֵּין לְכֹהֵן הֶדְיוֹט. וּמַמְזֶרֶת לְבֶן יִשְׂרָאֵל. וּבַת יִשְׂרָאֵל לְמַמְזֵר. וּבַת יִשְׂרָאֵל לְעַמּוֹנִי וּמוֹאָבִי. וּבַת יִשְׂרָאֵל לִפְצוּעַ דַּכָּא. וּכְרוּת שָׁפְכָה. וְהַנּוֹשֵׂא גְּרוּשָׁתוֹ אַחַר שֶׁנִּשֵּׂאת לְאַחֵר. וִיבָמָה שֶׁנִּשֵּׂאת לְזָר וַעֲדַיִין רְשׁוּת הַיָּבָם עָלֶיהָ. הַחֲלוּצָה הֲרֵי הִיא כַּגְּרוּשָׁה וְהִיא אֲסוּרָה לַכֹּהֵן מִדִּבְרֵי סוֹפְרִים. וְהַנְּתִינִים הֲרֵי הֵם כַּמַּמְזֵרִים אֶחָד זְכָרִים וְאֶחָד נְקֵבוֹת וְאִיסּוּרָם מִדִּבְרֵי סוֹפְרִים. וּבְהִלְכוֹת אִיסּוּרֵי בִּיאָה יִתְבָּאֵר לְךָ מַה הֵם הַנְּתִינִים:

@@Shoftim 32:2@@
וְיֵשׁ שֶׁאִסּוּר בִּיאָתָן בַּעֲשֵׂה וְאֵינוֹ מֵחַיָּיבֵי לָאוְיָן וּשְׁלֹשָׁה הֵם. מִצְרִי וַאֲדוֹמִי דּוֹר רִאשׁוֹן וְדוֹר שֵׁנִי אֶחָד זְכָרִים וְאֶחָד נְקֵבוֹת. וְהַבְּעוּלָה לְכֹהֵן גָּדוֹל. הֲרֵי שְׁלֹשָׁה שֶׁבִּיאָתָן הֵם בַּעֲשֵׂה וְלֹא בְּלָאו לְפִי שֶׁלֹּא נֶאֱמַר בְּאֵלּוּ לֹא יָבֹא אוֹ לֹא יִקַּח אֶלָּא מִכְּלָל שֶׁנֶּאֱמַר דּוֹר שְׁלִישִׁי יָבֹא לָהֶם בִּקְהַל ה'. אַתָּה לָמֵד שֶׁדּוֹר רִאשׁוֹן וְשֵׁנִי לֹא יָבֹא. וּמִכְּלַל שֶׁנֶּאֱמַר וְהוּא אִשָּׁה בִבְתוּלֶיהָ יִקָּח אַתָּה לָמֵד שֶׁאֵינָהּ בְּתוּלָה לֹא יִקַּח, וְלָאו הַבָּא מִכְּלַל עֲשֵׂה הֲרֵי הוּא כַּעֲשֵׂה, וכנז' כָּל זֶה בְּדִבְרֵי הָרַמְבַּ"ם זַ"ל הִלְכוֹת אִישׁוּת פ"א:

@@Shoftim 33:1@@
אָנוּס פָּטוּר מִכְּלוּם מִן הַמַּלְקוּת וּמִן הַקָּרְבָּן וְאֵין צָרִיךְ לוֹמַר מִן הַמִּיתָה שֶׁנֶּאֱמַר וְלַנַּעֲרָה לֹא תַעֲשֶׂה דָבָר (וְדָרְשׁוּ בְּסִפְרֵי מְלַמֵּד שֶׁפְּטָרָהּ הַכָּתוּב מִן הַמִּיתָה מִנַּיִן אַף מִן הַקָּרְבָּן ת"ל חָטָא מִנַּיִן אַף מִן הַמַּלְקוּת ת"ל חֵטְא מָוֶת מְלַמֵּד שֶׁכָּל אֳונָסִין שֶׁבַּתּוֹרָה פְּטוּרִין):

@@Shoftim 33:2@@
בַּמֶּה דְבָרִים אֲמוּרִים כְּשֶׁנֶּאֱנַס הַנִּבְעָל. אֲבָל הַבּוֹעֵל אֵין לוֹ אוֹנֶס שֶׁאֵין קִישּׁוּי אֶלָּא לְדַעַת, עכ"ל הָרַמְבַּ"ם זַ"ל. וְכָתַב הָרָאָבָ"ד זַ"ל יֵשׁ אוֹנֶס שֶׁהוּא פָּטוּר כְּגוֹן שֶׁהוּא מִתְכַּוֵּין לְאִשְׁתּוֹ וּתְקָפַתּוּ אַחַת מִן הָעֲרָיוֹת וֶהֱבִיאַתּוּ עָלֶיהָ. אוֹ שֶׁדְּבָקוּהוּ גּוֹיִם עָלֶיהָ עכ"ל. וְכָתַב הָרַב הַמַּגִּיד זַ"ל עַל דְּבָרָיו וְזַ"ל נִרְאֶה שֶׁבִּכְגוֹן זֶה אֲפִילּוּ רַבֵּינוּ (הָרַמְבַּ"ם) יוֹדֶה בּוֹ שֶׁכָּל שֶׁהַקּוֹשִׁי נַעֲשֵׂית לְדַעַת הֶתֵּר וְאַחַ"כּ נֶאֱנַס בִּהְיוֹתוֹ עוֹמֵד בְּקִשּׁוּיוֹ וַדַּאי אוֹנֶס גָּמוּר הוּא וְאֵין לוֹמַר עַל זֶה אֵין קִשּׁוּי אֶלָּא לְדַעַת עכ"ל:

@@Shoftim 33:3@@
וְכָתַב עוֹד הָרַב הַמַּגִּיד זַ"ל שָׁם עַל מ"ש רָבָא אֵין אוֹנֶס לַעֲרָיוֹת שֶׁאֵין קִישּׁוּי אֶלָּא לְדַעַת יֵשׁ מְפָרְשִׁים הַכַּוָּנָה שֶׁאִם אָמְרוּ לוֹ גּוֹיִם לָבֹא עַל הָעֶרְוָה וְאִם לָאו שֶׁיַּהַרְגוּ אוֹתוֹ וּבָא עָלֶיהָ הֲרֵי זוֹ מוּמָת עַל יָדָהּ. לְפִי שֶׁאֵין קִישּׁוּי אֶלָּא לְדַעַת, וְזֶה דַּעַת רַבֵּינוּ (הָרַמְבַּ"ם) אֲבָל יֵשׁ אוֹמְרִים דְּכֵיוָן שֶׁמֵּחֲמַת הָאוֹנֶס יְהַב דַּעְתֵּיהּ וְנִתְקַשָּׁה אַף הוּא פָּטוּר בְּדִינֵי אָדָם. וְהָא דְּרָבָא מַשְׁכַּחַת לָהּ כְּגוֹן שֶׁתְּקָפַתּוּ אִשָּׁה וְקִרְבָתוֹ לְעַצְמָהּ לְאָונְסוֹ אוֹ שֶׁדָּבְקוּ אוֹתוֹ גּוֹיִם לָעֶרְוָה. וְאֵין שָׁם פַּחַד מִיתָה כֵּיוָן דְּלֵיכָּא אוֹנֶס דְּגוּפֵיהּ, אע"פ שֶׁדְּבֵקִים זֶה לָזֶה וּבָא עָלֶיהָ חַיָּיב שֶׁאִלְמָלֵא נָתַן דַּעְתּוֹ עָלֶיהָ לֹא הָיָה אֶפְשָׁר לוֹ לִבְעוֹל שֶׁאֵין קִישּׁוּי אֶלָּא לְדַעַת וְכוּ' עכ"ל. וְיֵשׁ לְדַבֵּר בְּכָל הָאָמוּר ואכמ"ל:

@@Shoftim 34:1@@
הָאִשָּׁה שֶׁתְּחִלַּת בִּיאָתָהּ בְּאוֹנֶס וְסוֹפָהּ בְּרָצוֹן פְּטוּרָה מִכְּלוּם, שֶׁמִּשֶּׁהִתְחִיל לִבְעוֹל בְּאוֹנֶס אֵין בְּיָדָהּ שֶׁלֹּא תִּרְצֶה שֶׁיֵּצֶר הָאָדָם וְטִבְעוֹ כּוֹפֶה אוֹתָהּ לִרְצוֹת. הָרַמְבַּ"ם בָּהּ' א"ב פֶּרֶק א' ה"ט:

@@Shoftim 35:1@@

אַחַ"כּ יִתְפּוֹס בְּיָדוֹ אַרְבָּעָה גֵּרוּשׁ אוֹ יוֹתֵר לְהַפְרִישָׁם לִצְדָקָה וְיֹאמַר בַּקָּשָׁה זוֹ יהר"מ ה' או"א שֶׁתַּעֲשֶׂה לְמַעַן רַחֲמֶיךָ וַחֲסָדֶיךָ וּלְמַעַן פְּסוּקִים וְסָתְרִי תּוֹרָה שֶׁקָּרִיתִי לְפָנֶיךָ וּלְמַעַן מִצְוַת צְדָקָה זוֹ שֶׁאֲנִי מַפְרִישׁ בְּיָדִי. וְתָחוֹס וְתַחְמוֹל וּתְרַחֵם עָלַי לְתַקֵּן אֲשֶׁר פִּגַּמְתִּי בְּהִסְתַּכְּלוּתִי בִּמְקוֹם עֶרְוָה שֶׁל אִשָּׁה שֶׁהִיא אֲסוּרָה עָלַי בְּאִיסּוּר עֲרָיוֹת. וַהֲרֵינִי מִתְחָרֵט עַל אֲשֶׁר חָטָאתִי בְּעָוֹן זֶה וְעוֹזֵב אֲנִי חֵטְא זֶה עֲזִיבָה גְּמוּרָה וּבְרוֹב חֲסָדֶיךָ הַגְּדוֹלִים תְּרַחֵם עָלַי שֶׁלֹּא אֶתְגַּלְגֵּל בָּעוֹף הַנִּקְרָא רָאָה וּתְנַחֲלֵנִי חֵלֶק טוֹב לָעוֹלָם הַבָּא וְיַעֲלֶה לְפָנֶיךָ כְּאִלּוּ כִּוַּנְתִּי בְּכָל הַכַּוָּנוֹת הָרְאוּיוֹת לְכַוֵּן וַיְהִי נוֹעַם ה' וְכוּ' ב"פ יִהְיוּ לְרָצוֹן אִמְרֵי פִי וְכוּ':
