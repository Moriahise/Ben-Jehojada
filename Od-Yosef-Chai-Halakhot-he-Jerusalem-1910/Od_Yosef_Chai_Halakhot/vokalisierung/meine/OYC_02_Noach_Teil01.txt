# Meine Fassung · Vorlage aus Nakdan (ohne Dicta-Plene-Marken) · hier korrigieren
@@Noach 1:1@@
בזה"ז דְּלֵיכָּא תְּכֵלֶת וְעוֹשִׂין כָּל הֵד' חוּטִין מִלָּבָן, הִנֵּה בָּזֶה מְקַיְּימִים מ"ע שֶׁל צִיצִית כְּתִקְנָהּ, דקיי"ל כַּחֲכָמִים דס"ל אֵין הַתְּכֵלֶת מְעַכֵּב אֶת הַלָּבָן וְאֵין הַלָּבָן מְעַכֵּב אֶת הַתְּכֵלֶת, ואע"ג דְּהָרָזָ"ה זַ"ל ס"ל כמ"ד הַתְּכֵלֶת מְעַכֵּב אֶת הַלָּבָן, וְכֵן הוּא סְבָרַת רַבּוֹ שֶׁל מָהָרָ"ם בַּר בָּרוּךְ זַ"ל, מ"מ כָּל הַפּוֹסְקִים פֶּה אַחַת שָׂפָה אַחַת פָּסְקוּ כַּחֲכָמִים, וְאֵין לָחוּשׁ לִסְבָרַת הנז', וְיוּכַל לָצֵאת לרה"ר בזה"ז בְּטַלִּית מְצוּיֶּיצֶת וְלֵית לָן בָּהּ, וכמ"ש בְּלֵב חַיִּים סִי' ע"ג ע"ש:

@@Noach 2:1@@
מִצְנֶפֶת אע"פ שֶׁיֵּשׁ לָהּ ד' כְּנָפוֹת פְּטוּרָה, שֶׁנֶּאֱמַר כְּסוּתְךָ אֲשֶׁר תְּכַסֶּה בָּהּ, פְּרָט לַמִּצְנֶפֶת שֶׁאֵינָהּ עֲשׂוּיָה לְכַסּוֹת הַגּוּף אֶלָּא אֶת הָרֹאשׁ בִּלְבַד, וַאֲפִילּוּ בְּמָקוֹם שֶׁנּוֹהֲגִין לִהְיוֹת רָאשֵׁי הַמִּצְנֶפֶת מוּשְׁלָכִים עַל כְּתֵפָם וְגוּפָם וּמִתְכַּסֶּה בָּהּ רֹאשָׁם וְרוּבָּם פְּטוּרָה, כֵּיוָן שֶׁעִיקָּרָהּ לְכַסּוֹת הָרֹאשׁ עֲשׂוּיָה:

@@Noach 3:1@@
הַחוּטִין שֶׁל הַצִּיצִיּוֹת צְרִיכִין שֶׁיִּהְיוּ טוֹוִין לִשְׁמָן, שֶׁיֹּאמַר בְּפִיו בִּתְחִלַּת הַטְּוִוי שֶׁהוּא טוֹוֶה לְשֵׁם צִיצִית, וְיָכוֹל לוֹמַר בְּכָל לָשׁוֹן שֶׁיּוֹדֵעַ, וְאִם לֹא טוֹוָה לִשְׁמָן פְּסוּלִין, שֶׁנֶּאֱמַר תַּעֲשֶׂה לְךָ לְשֵׁם חוֹבְךָ, וְאִם טוֹוָם גּוֹי אע"פ שֶׁיִּשְׂרָאֵל עוֹמֵד עַל גַּבָּיו וְאוֹמֵר שֶׁיַּעֲשֵׂם לְשֵׁם צִיצִית הֲרֵי אֵלּוּ פְּסוּלִין, אַךְ בִּשְׁעַת הַדְּחָק שֶׁאֵין לוֹ מִי שֶׁיִּטְוֶוה לִשְׁמָן לֹא אִישׁ וְלֹא אִשָּׁה יִשְׂרְאֵלִית כִּי אִם גּוֹי יֹאמַר לְגוֹי שֶׁיִּטְוֶוה לוֹ הַחוּטִין לְשֵׁם צִיצִית וְהַיִּשְׂרָאֵל וְסִיַּיע ג"כ קְצָת בִּטְוִויָּה, וְגַם יַעֲמוֹד הַיִּשְׂרָאֵל, אֵצֶל הַטּוֹוֶה מִתְּחִלָּה וע"ס, וִיכַוֵּין יִשְׂרָאֵל לְשֵׁם צִיצִית, וְאַפֵּ"ה כַּאֲשֶׁר יִמְצָא לוֹ אַחַ"כּ צִיצִיּוֹת שֶׁטּוֹוֶה אוֹתָם יִשְׂרָאֵל יַחֲלִיף צִיצִיּוֹת אֵלּוּ:

@@Noach 4:1@@
חוּטֵי צִיצִית צְרִיכִין שְׁזִירָה, שֶׁנֶּאֱמַר פְּתִיל תְּכֵלֶת, שֶׁיִּהְיֶה טָווּי וְשָׁזוּר כְּעֵין פְּתִילָה, וְלָכֵן צָרִיךְ שֶׁגַּם הַשְּׁזִירָה יַעֲשׂוּ אוֹתָהּ לְשֵׁם צִיצִית, שֶׁיֹּאמַר אֲנִי שׁוֹזֵר חוּטִין אֵלּוּ לְשֵׁם צִיצִית, וְיֹאמַר בְּכָל לָשׁוֹן שֶׁיּוֹדֵעַ, וְאִם לֹא הָיוּ שֶׁזּוֹרִין לִשְׁמָן פְּסוּלִים וְיֵשׁ בַּעֲלֵי נֶפֶשׁ שֶׁמַּחְמִירִין עַל עַצְמָן אֲפִילּוּ בְּנִיפּוּץ לַעֲשׂוֹתוֹ לְשֵׁם צִיצִית, אַךְ לֹא נָהֲגוּ הָעוֹלָם בָּזֶה, וכנז' בָּאַחֲרוֹנִים זַ"ל:

@@Noach 5:1@@
אִם הָאִשָּׁה הִיא טוֹוָה וְשׁוֹזֶרֶת יֹאמַר לָהּ תְּחִלָּה טְוִי לִי צִיצִית לְטַלִּית וְכֵן שְׁזוּרִי לִי צִיצִית לְטַלִּית, וְיַנִּיחַ אוֹתָהּ ג"כ שֶׁתֹּאמַר הִיא בִּתְחִלַּת הַטְּוִויָּה וְהַשְּׁזִירָה שֶׁהִיא טוֹוָה וְשׁוֹזֶרֶת לְשֵׁם צִיצִית, וְיֵשׁ לִיזָּהֵר שֶׁלֹּא יִטְווּם וְיִשְׁזְרֵם קָטָן אוֹ קְטַנָּה. גַּם צָרִיךְ לִיזָּהֵר מְאֹד שֶׁלֹּא יִקְנֶה צִיצִיּוֹת מִכָּל אָדָם, אֶלָּא רַק מֵאוֹתָם שֶׁהֵם מוּחְזָקִים לְנֶאֱמָנִים וְיִרְאֵי הַשֵּׁם, שֶׁיּוֹדֵעַ בָּהֶם שֶׁבְּקִיאִין לַעֲשׂוֹתָם לִשְׁמָן כְּדֵי שֶׁלֹּא תִּהְיֶה בִּרְכָתוֹ לְבַטָּלָה, וְגַם לֹא יְבַטֵּל מִצְוַת צִיצִית, וכנז' בחס"ל:

@@Noach 6:1@@
הַתּוֹרָה אָמְרָה צִיצִית הַכָּנָף שֶׁיִּהְיוּ הַצִּיצִיּוֹת מִדְּבָרִים שֶׁעוֹשִׂים מֵהֶם הַכָּנָף שֶׁהוּא הַבֶּגֶד, וְלָכֵן כָּל מִינֵי צֶמֶר הַגֵּרוּעִין שֶׁאֵין בֶּגֶד נַעֲשֵׂית מֵהֶם פְּסוּלִין לְצִיצִית, מִטַּעַם הָאָמוּר, וּלְכָךְ אֵין לוֹקְחִים צִיצִית אֶלָּא מֵאָדָם נֶאֱמָן, וְצָרִיךְ שֶׁיַּכִּיר בּוֹ שֶׁהוּא יוֹדֵעַ הִלְכוֹת הַפּוֹסְלִים בַּצִּיצִית וְנִזְהָר מֵהֶם, כִּי אוּלַי הוּא עַם הָאָרֶץ וְנִזְדַּמֵּן שֶׁעָשָׂה הַצִּיצִיּוֹת מִן פְּסוֹלֶת הַצֶּמֶר מִפְּנֵי שֶׁאֵינוֹ יוֹדֵעַ הַדִּין:

@@Noach 6:2@@
וְנוֹדַע דַּרְכָּם שֶׁל שׁוֹחֲטֵי מָתָא שֶׁתּוֹלֵשׁ הַשּׁוֹחֵט צֶמֶר מִן צַוַּאר הַבְּהֵמָה בִּמְקוֹם שְׁחִיטָה כְּדֵי שֶׁיִּשְׁחוֹט יָפֶה, וְהַמִּנְהָג שֶׁזֶּה הַצֶּמֶר שֶׁתּוֹלֵשׁ הַשּׁוֹחֵט מִצַּוַּאר הַבְּהֵמָה הוּא שֶׁלּוֹ, וּמֵחֲמַת שֶׁשּׁוֹחֵט בְּהֵמוֹת הַרְבֵּה יִהְיֶה מְקוּבָּץ אֶצְלוֹ מִזֶּה הַצֶּמֶר הַרְבֵּה וּמוֹכְרוֹ וְהַמָּעוֹת שֶׁלּוֹ, וְאֵין לְבַעֲלֵי הַבְּהֵמוֹת עִרְעוּר בָּזֶה, וְאֶפְשָׁר דְּצֶמֶר זֶה דִּינוֹ כְּצֶמֶר הַנֶּאֱחָז בַּקּוֹצִין כְּשֶׁהַצֹּאן רוֹבְצִים בֵּינֵיהֶם דקי"ל דְּפָסוּל זֶה לְצִיצִית, וְגַם דָּמֵי לְנִימִין הַנִּתְלָשִׁים מִן הַצֹּאן וְשִׁיּוּרֵי שְׁתִי שֶׁהָאוֹרֵג מְשַׁיֵּיר בְּסוֹף הַבֶּגֶד, דְּכָל אֵלּוּ הֵם פְּסוּלִין לְצִיצִית דַּחֲשִׁיבִי פְּסוֹלֶת, וְכֵן נָמֵי הה"ד בְּצֶמֶר שֶׁתּוֹלֵשׁ מִן הַבְּהֵמָה בִּשְׁעַת שְׁחִיטָה, וְלָכֵן לְכַתְּחִלָּה לֹא יַעֲשֶׂה מִזֶּה צִיצִיּוֹת:

@@Noach 7:1@@
גָּזַל חוּטִין שֶׁנִּטְווּ לְשֵׁם צִיצִית וְהָלַךְ הַגַּזְלָן וְעָשָׂה מֵהֶם צִיצִית הֲרֵי הֵם פְּסוּלִים. וְאִם מְכָרָן הַגַּזְלָן לְאַחֵר, אִם כְּבָר נִתְיָיאֲשׁוּ הַבְּעָלִים הֲרֵי אֵלּוּ כְּשֵׁרִים מִפְּנֵי שֶׁקְּנָאָם הַלּוֹקֵחַ בְּיֵאוּשׁ וְשִׁינּוּי רְשׁוּת מֵהַגַּזְלָן לַלּוֹקֵחַ, וְעַיֵּין בש"ע לר"ז זַ"ל ע"ש וְטוֹב לְהַחְמִיר בָּזֶה:

@@Noach 8:1@@
גָּזַל צֶמֶר שֶׁלֹּא נִטְוָוה וְטַוָּואָן הַגַּזְלָן לְשֵׁם צִיצִית, אִם עֲדַיִין לֹא נִתְיָיאֲשׁוּ הַבְּעָלִים מִמֶּנּוּ הֲרֵי אֵלּוּ פְּסוּלִים לְצִיצִית, מִפְּנֵי דַּעֲדַיִין שֵׁם גֶּזֶל נִקְרָא עֲלֵיהֶם, וְלֹא קְנָאָם הַגַּזְלָן בְּשִׁינּוּי זֶה שֶׁעֲשָׂאָם חוּטִין, מִפְּנֵי שֶׁשִּׁינּוּי זֶה חוֹזֵר לִבְרִיָּיתוֹ, דְּאֶפְשָׁר לַעֲשׂוֹת אֶת הַחוּטִין צֶמֶר כִּדְהָווּ מֵעִיקָּרָא. אֲבָל אִם נִתְיָיאֲשׁוּ הַבְּעָלִים אוֹ אֲפִילּוּ לֹא נִתְיָיאֲשׁוּ אֶלָּא שֶׁעָשָׂה בָּהֶם הַגַּזְלָן שִׁינּוּי גָּמוּר שֶׁאֵינוֹ חוֹזֵר לִבְרִיָּיתוֹ, כְּגוֹן שֶׁלִּיבֵּן אֶת הַצֶּמֶר בְּגָפְרִית וע"ד שֶׁנִּתְבָּאֵר בח"מ סִי' ש"ס, הִנֵּה כֵּיוָן שֶׁבָּזֶה קָנָה הַגַּזְלָן אֶת הַצֶּמֶר לְגַמְרֵי, שֶׁאֲפִילּוּ אִם הַצֶּמֶר בְּעַיִן אֵינוֹ צָרִיךְ לְהַחֲזִיר לְהַנִּגְזָל אֶלָּא רַק דְּמֵי הַצֶּמֶר, א"כ הֲרֵי הוּא חֲשִׁיב שֶׁלּוֹ מַמָּשׁ לְכָל דָּבָר, לָכֵן הַצִּיצִית שֶׁעָשָׂה מִמֶּנּוּ כְּשֵׁרִים הֵם, אֲבָל מ"מ לֹא יְבָרֵךְ עֲלֵיהֶם, דְּאֵין זֶה מְבָרֵךְ אֶלָּא מְנָאֵץ כֵּיוָן דע"י גְּזֵילָה בָּא הַצִּיצִית לְיָדוֹ, וכמ"ש הַגָּאוֹן ר"ז זַ"ל בש"ע:

@@Noach 9:1@@
יַעֲשֶׂה הַצִּיצִיּוֹת י"ב גּוּדָלִים, וְכַאֲשֶׁר כָּתַבְתִּי בסה"ק בֶּן אִישׁ חַי בְּפָרָשָׁה זוֹ ע"ש, וְאִם הוּא פָּחוֹת מִן י"ב גּוּדָלִים, וכ"ש אִם הָיָה בִּתְחִלָּה כְּשִׁיעוּר זֶה וְאַחַ"כּ נִתְמַעֵט ע"י רְחִיצָה שֶׁנִּתְכַּוְּוצוּ כְּשֵׁרִים בְּדִיעֲבַד, וכנז' בְּלֵב חַיִּים סִי' ע"ט בְּאוֹרֶךְ:

@@Noach 10:1@@

@@Noach 13:2@@
