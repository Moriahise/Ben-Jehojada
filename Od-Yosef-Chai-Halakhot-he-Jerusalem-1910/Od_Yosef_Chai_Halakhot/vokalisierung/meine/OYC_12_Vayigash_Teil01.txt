@@Vayigash 8:8@@
גַּם ס"ת פּוֹתֵחַ אֶת יָדֶיךָ חת"ך וְהוּא שֵׁם קָדוֹשׁ הַמְכוּנֶּה בְּסִפְרֵי הַמְּקוּבָּלִים שֵׁם הַפַּרְנָסָה, גַּם תְּכַוֵּין כִּי שֵׁם הֲנֵז' בְּחִלּוּף א"ת ב"ש הוּא סָאָ"ל, וְהוּא שֵׁם אֶחָד מִן ע"ב שֶׁל וַיִּסַּע וַיָּבֹא וַיֵּט, גַּם שֵׁם סָאָ"ל הוּא בְּגֵימַטְרְיָא שֵׁם פא"י כְּמִנְיַן יאהדונה"י כַּנָּז' בר"ת פא"י:

@@Vayigash 8:9@@
וְהִנֵּה כָּל הַכַּוָּנוֹת שֶׁבַּפָּסוּק הַזֶּה הֵם שָׁווֹת בִּשְׁלֹשָׁה זְמַנִּים שֶׁאוֹמְרִים אַשְׁרֵי. וְהַחִילּוּק שֶׁבֵּינֵיהֶם אֵינוֹ אֶלָּא בְּכַוָּנַת שֵׁם חת"ך בִּלְבַד. וְזֶה עִנְיָינוֹ כִּי בִּזְמִירוֹת דְּשַׁחֲרִית תְּכַוֵּין כִּי שֵׁם חת"ך שָׁרְשׁוֹ הוּא יוֹצֵא מֵחִילּוּף וּתְמוּרַת ההוי"ה שֶׁהִיא בַּגַּי' מצפ"ץ, כִּי הוי"ה בא"ת ב"ש הוּא מצפ"ץ שֶׁהוּא בַּגַּי' ש'. וּכְבָר נִתְבָּאֵר במ"א כִּי שֵׁם מצפ"ץ לִהְיוֹתוֹ תְּמוּרַת ההוי"ה לָכֵן הוּא רַחֲמִים מְעוֹרָב בַּדִּין, וְלָכֵן הוּא בַּגַּי' אֱלֹהִים דיודי"ן וְזֶה סִדְרָן הוי"ה וְאַחַ"כּ מצפ"ץ וְאַחַ"כּ אֱלֹהִים כִּי שֵׁם אלהי"ם הוּא יוֹתֵר דִּין מִכּוּלָּם. וְהִנֵּה בִּזְמִירוֹת שֶׁעֲדַיִין לֹא הִתְפַּלַּלְנוּ תְּפִלַּת הַשַּׁחַר וְלֹא יָרַד שֶׁפַע עֶלְיוֹן אֶלָּא שעכ"ז הוּא בְּחִינַת בֹּקֶר אֲשֶׁר בּוֹ שׁוֹלֵט אַבְרָהָם אִישׁ הַחֶסֶד, לָכֵן שֵׁם מצפ"ץ שֶׁהוּא מְמוּצָּע בַּדִּין הוּא הַמֵּאִיר עַתָּה. וְנִמְצָא כִּי שֵׁם זֶה הַנִּקְרָא חת"ך עִם הֱיוֹת שֶׁמְּקוֹרוֹ הוּא מִשְּׁלֹשָׁה שֵׁמוֹת שֶׁהֵם הוי"ה מצפ"ץ אֱלֹהִים (שֶׁמִּסְפָּרָם חת"ך) כ"ז כָּל אֶחָד הוּא כְּפִי הַזְּמַן וְהַשָּׁעָה, וְעַתָּה בִּזְמַן הַזְּמִירוֹת הִנֵּה הַשֵּׁם הַזֶּה יוֹנֵק מִשֵּׁם מצפ"ץ הַמְמוּצָּע וְכוּ':

@@Vayigash 8:10@@
וְכַוָּנַת שֵׁם חת"ך בְּאַשְׁרֵי שֶׁקּוֹדֶם וּבָא לְצִיּוֹן אֲשֶׁר אָז כְּבָר הִתְפַּלַּלְנוּ תְּפִלַּת שַׁחֲרִית וְיֵרֵד הַשֶּׁפַע וְגַם הוּא בְּתִינַת שֶׁהוּא חֶסֶד גָּמוּר אָז הוּא נִגְזַר מִשֵּׁם ההוי"ה שֶׁהוּא רַחֲמִים וְהוּא הוי"ה דע"ב דיודי"ן. וְהִנֵּה נוֹדַע בְּסוֹד וְהַחָכְמָה תִּחְיֶה בְּעָלֶיהָ כִּי כָּל הַמָּזוֹן וְהַפַּרְנָסָה מִן הַחָכְמָה כנז' בַּזּוֹהַר פ' בְּשַׁלַּח בְּעִנְיַן הַמָּן. וה"ס פּוֹתֵחַ אֶת יָדֶיךָ שֶׁהוּא יודי"ך שֶׁהֵם ד' יודי"ן שבהוי"ה דע"ב דיודי"ן כַּנָּז' וְכָל אֶחָד כָּלוּל. מֵעֶשֶׂר וְעֶשֶׂר מַעֲשֵׂר הֲרֵי ד' יודי"ן הֵם ת' וכ"ח אוֹתִיּוֹת שֶׁיֵּשׁ בְּמִלּוּי הַמִּלּוּי שֶׁיֵּשׁ בהוי"ה זוֹ הֲרֵי הַכֹּל חת"ך, וּבָזֶה תָּבִין הֱיוֹת הַשֵּׁם הַזֶּה נִקְרָא אֵצֶל הַמְקוּבָּלִים שֵׁם הַפַּרְנָסָה:

@@Vayigash 8:11@@
אֲבָל כַּוָּנַת שֵׁם חת"ך בִּזְמַן הַמִּנְחָה הוּא נִגְזַר מִשֵּׁם אֱלֹהִים לְפִי שֶׁאָז הוּא זְמַן הַדִּין וַהֲרֵי נִתְבָּאֵר שְׁלֹשָׁה חִילּוּקִים שֶׁיֵּשׁ בְּכַוָּנַת שֵׁם חת"ך בִּשְׁלֹשָׁה זְמַנִּים הֲנֵז' אֲבָל כָּל שְׁאָר הַכַּוָּנוֹת שֶׁבַּפָּסוּק הַזֶּה הֵם שָׁוִים בִּשְׁלֹשָׁה זְמַנִּים הֲנֵז':

@@Vayigash 8:12@@
וְהִנֵּה בְּמִלַּת וּמַשְׂבִּיעַ ג"כ תְּכַוֵּין שֶׁהוּא בְּגֵימַטְרְיָא חת"ך הַנִּזְכָּר וּתְכַוֵּין ג"כ בּוֹ כְּמִנְיַן כַּוָּנוֹת שֵׁם חָתַךְ הֲנֵז'. וּבְמִלַּת רָצוֹן תְּכַוֵּין לְהַמְשִׁיךְ מִן הָרָצוֹן הָעֶלְיוֹן שֶׁהוּא הוי"ה הָעוֹלֶה קפ"ד שֶׁהוּא לִימּוּד דְּחָכְמָה וְגַם הוּא אֶהְיֶה דיודי"ן הָעוֹלֶה בַּגַּי' קס"א שֶׁהוּא הַיְּסוֹד דְּאִימָּא, וְהִנֵּה קפ"ד וקס"א עִם הַכּוֹלֵל הֵם בַּגַּי' רָצוֹן וּתְכַוֵּין לְהַמְשִׁיךְ מִשְּׁנֵי יְסוֹדוֹת הֲנֵז' שֶׁפַע וּמָזוֹן וּבְרָכָה אֶל כָּל הָעוֹלָם:
