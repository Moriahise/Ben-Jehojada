לָא מִבָּעֲיָא דְּאָסוּר לְכֹהֵן לִיכָּנֵס לְתוֹךְ חֶדֶר שֶׁהַמֵּת מוּנָּח בּוֹ אֶלָּא גַּם לְחֶדֶר אוֹ עֲלִיָּה הַפְּתוּחִים לְאוֹתוֹ בַּיִת שֶׁהַמֵּת מוּנָּח בּוֹ ג"כ אָסוּר לִיכָּנֵס שָׁם. וְכֹל שֶׁיֵּשׁ מִזֶּה לָזֶה נֶקֶב פָּתוּחַ שֶׁהוּא טֶפַח עַל טֶפַח אָסוּר לִיכָּנֵס שָׁם, וְכֵן אֲפִילּוּ כַּמָּה חֲדָרִים פְּתוּחִים זֶה לָזֶה אָסוּר לִיכָּנֵס בְּתוֹךְ כּוּלָּם. אֲבָל אִם סָתַם הַנֶּקֶב הַפָּתוּחַ מֵחֶדֶר לְחֶדֶר מוּתָּר לִיכָּנֵס בַּחֶדֶר שֶׁבּוֹ הַסְּתִימָה כִּי הַסְּתִימָה חוֹצֶצֶת וְאֵין הַטּוּמְאָה נִכְנֶסֶת שָׁם. ואע"פ שֶׁסְּתָמוֹ לְפִי שָׁעָה וְאֵין בְּדַעְתּוֹ שֶׁתִּשָּׁאֵר שָׁם לְעוֹלָם אַפֵּ"ה חוֹצֶצֶת. מִיהוּ אִם סְתָם בְּדָבָר שֶׁמְּקַבֵּל טוּמְאָה אֵינוֹ חוֹצֵץ בִּפְנֵי הַטּוּמְאָה:

@@Tetzaveh 10:1@@
אֲפִילּוּ אִם הָיְתָה הַסְּתִימָה בְּסָרוּג וּקְלִיעָה שֶׁכּוּלָּהּ נְקָבִים אִם אֵין בַּנְּקָבִים שֶׁל הַסְּרִיגָה פּוֹתֵחַ טֶפַח בְּשׁוּם אֶחָד מֵהֶם הֲרֵי זֶה חוֹצֵץ וְאֵין נִקְבֵי הַסְּרִיגוֹת מִצְטָרְפִים לַפּוֹתֵחַ טֶפַח:

@@Tetzaveh 10:2@@
וְאָסוּר לְכֹהֵן לְהִתְקָרֵב תּוֹךְ ד' אַמּוֹת שֶׁל מֵת אוֹ תּוֹךְ ד' אַמּוֹת שֶׁל קֶבֶר. אֲבָל אִם יֵשׁ מְחִיצּוֹת גְּבוֹהוֹת עֲשָׂרָה טְפָחִים סְבִיב הַקֶּבֶר אוֹ שֶׁהַקֶּבֶר מוּנָּח בְּחָרִיץ עָמוֹק עֲשָׂרָה טְפָחִים אֵין צָרִיךְ לְהִתְרַחֵק מִמֶּנּוּ אֶלָּא רַק אַרְבָּעָה טְפָחִים וכנז' בש"ע סִי' שמ"א:

@@Tetzaveh 11:1@@
מָבוֹי שֶׁיֵּשׁ בּוֹ כַּמָּה חֲצֵרוֹת וּבְרֹאשׁ הַמָּבוֹי יֵשׁ שְׁתֵּי מְזוּזוֹת וּמַשְׁקוֹף וְהַמֵּת בְּחָצֵר אֶחָד מִן הַחֲצֵרוֹת שֶׁבְּוַדַּאי מוֹצִיאִין אוֹתוֹ מִפֶּתַח הַמָּבוֹי, אָסוּר לְכֹהֵן לִיכָּנֵס אֶל הַמָּבוֹי אוֹ לָצֵאת מִמֶּנּוּ אֲפִילּוּ אִם יֵשׁ בֵּית הַכְּנֶסֶת בְּאוֹתוֹ מָבוֹי וְלֹא יֵשׁ בֵּית הַכְּנֶסֶת אַחֶרֶת בָּעִיר כִּי אִם בְּאוֹתוֹ מָבוֹי וְרוֹצֶה לִיכָּנֵס לְהִתְפַּלֵּל עִם הַצִּיבּוּר וְלַעֲנוֹת קַדִּישׁ וּקְדוּשָּׁה וּבָרְכוּ וְגַם לוֹמַר בִּרְכַּת כֹּהֲנִים אַפֵּ"ה אָסוּר לְהִכָּנֵס אוֹ לָצֵאת דֶּרֶךְ אוֹתוֹ הַפֶּתַח מִפְּנֵי כִּי טוּמְאָה דַּרְכָּהּ לָצֵאת מִשָּׁם מִתַּחַת הַמַּשְׁקוֹף שֶׁהוּא מַאֲהִיל שָׁם:

@@Tetzaveh 11:2@@
בד"א בִּימֵי הַחוֹל אוֹ אֲפִילּוּ ביו"ט שֶׁקּוֹבְרִין הַמֵּת בּוֹ בַּיּוֹם וְסוֹף הַטּוּמְאָה לָצֵאת דֶּרֶךְ אוֹתוֹ הַפֶּתַח שֶׁל הַמָּבוֹי, אֲבָל בְּשַׁבָּת כֵּיוָן דְּאֵין קוֹבְרִין אוֹתוֹ בּוֹ בַּיּוֹם אֶלָּא עַד הַלַּיְלָה מוּתָּר לְכֹהֵן לִיכָּנֵס דֶּרֶךְ אוֹתוֹ הַפֶּתַח וְלָצֵאת מִמֶּנּוּ וְרַק אַחַר חֲשֵׁיכָה אָסוּר לְכֹהֵן לַעֲבוֹר דֶּרֶךְ אוֹתוֹ הַפֶּתַח שֶׁל הַמָּבוֹי:

@@Tetzaveh 12:1@@
אע"פ דְּנָכוֹן לְהִזָּהֵר הַכֹּהֵן שֶׁלֹּא יֵלֵךְ עַל קִבְרֵי גּוֹיִם עכ"ז בִּשְׁבִיל צוֹרֶךְ מִצְוָה כְּגוֹן לָלֶכֶת לְבֵית הַכְּנֶסֶת אוֹ לְבֵית הַמִּדְרָשׁ וְכַיּוֹצֵא מוּתָּר לוֹ לֵילֵךְ אִם יֵשׁ לְפִי דַּרְכּוֹ קִבְרֵי גּוֹיִם, וְלָכֵן פֹּה עִירֵנוּ בַּגְדָאד יע"א כְּשֶׁהוֹלְכִים לְהִשְׁתַּטֵּחַ עַל מַצֶּבֶת אֲדוֹנֵינוּ יְהוֹשֻׁעַ כה"ג אע"פ שֶׁיֵּשׁ בְּאֶמְצַע הַדֶּרֶךְ קִבְרֵי גּוֹיִם הוֹלְכִים הַכֹּהֲנִים בֵּינֵיהֶם וְאֵין חוֹשְׁשִׁין וְכֵן הַמִּנְהָג פָּשׁוּט כֵּיוָן דְּזֶה חֲשִׁיב הֲלִיכָה שֶׁל מִצְוָה:
